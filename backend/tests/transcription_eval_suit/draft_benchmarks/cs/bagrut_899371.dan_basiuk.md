found = true ;
}
}

public static void ArrangeMiror (int[]arr)
{
int[] New = new int (arr.Length) ; // המערך החדש
int NewCount = 0 ;

for (int i = 0 ; i < arr.Length ; i ++ )
{
if (arr[i] > 0)
{
New[NewCount] = arr[i] ;
New[NewCount + 1] = -arr[i] ;
NewCount = NewCount + 2 ;
}
}
Return New ;
}
