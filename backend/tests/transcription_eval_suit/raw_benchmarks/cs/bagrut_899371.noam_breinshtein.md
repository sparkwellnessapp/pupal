public static bool isMirror (int[] arr)
{
int count = 0;
for (int i = 0; i < arr.length; i++)
{
count = 0;
for (int j = 0; j < arr.length; j++)
{
if (arr[i] != 0 && arr[i] == (arr[j] - -1))
count ++;
}
if (count != 1)
return false;
}
return true;
}

public static void ArrangeMirror(int[] arr)      
{
int[] newArr = new int[arr.length]; // מניח שזה כבר מאופס

for (int i=0; i<arr.length; i++)
{
if (arr[i]<0)

arr[i]=0;
}

for (int i=0; i<arr.length; i+=2)
{
for (int j=0; j<arr.length; j++)
{
if (arr[j]!=0)
{
newArr[i] = arr[j];

newArr[i+1] = arr[j]*-1;
arr[j] = 0;
j += arr.length;
}
}

return newArr;
}
