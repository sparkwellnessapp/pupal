Public static int [] DiceStatictics (int [] arr)
{
    int [] cube = new int [21] ; // מערך המכיל כמות הפעמים המופיע
                                 // ערך במערך
    For (int i=0 ; i<cube.length ; i++) // איפוס מערך
    {
        cube[i] = 0 ;
    }

    For (int i=0 ; i<arr.length ; i++) // מונה כמות פעמים של ערך מסוים
    {
        cube[arr[i]] ++ ;
    }
    return cube ; // החזרת המערך
}

Public static void PrintStatistics (int[]arr)

                             // בהטלות

    }

Public bool HandleNewWorkShop(workShop ws)
{
    for (int i = 0 ; i < countWorkShops ; i++)
    {
        if (workShop[i].IsSimilarWorkShop(ws) == true) // אם סדנה במערך שקיימת
                                                       // דומה לסדנה שקיבלנו
        {
            if ((workShop[i].GetPeople() + ws.GetPeople()) <= 40) // בודקים אם יש מקום
            {
                workShop[i].GetPeople() += ws.GetPeople() ; // אם יש מקום מוסיפים את מספר האנשים מהסדנה החדשה לקיימת
                return true ;
            }
        }
    }

    if (countWorkShops < 200) // לא נמצאה סדנה דומה בדיקה אם יש מקום
                              // לסדנה חדשה
    {
        workShop[countWorkShop] = ws ; // יש הוספה
        countWorkShops ++ ; // קידום מספר התא
        return true ;
    }

    return false ; // לא נמצא מקום בסדנה קיימת אין מקום במערך
}
