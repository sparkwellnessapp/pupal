public static int[] DiceStatistics(int[] arr)
{
int[] counters = new int[21];

for (int i = 0; i < counters.Length; i++)
{
counters[i] = 0;
}
for (int i = 0; i < arr.Length; i++)
{
for (int j = 1; j < counters.Length; j++)
{
if (j == arr[i])
counters[j]++;
}
}
return counters;
}
